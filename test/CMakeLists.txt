include(CheckCXXCompilerFlag)

set(CXX11_STANDARD c++11)

if(SCOPE_GUARD_MSVC_FRONTEND)
    set(OPTIONS /W4 /WX)
    check_cxx_compiler_flag(/permissive- HAS_PERMISSIVE_FLAG)
    if(HAS_PERMISSIVE_FLAG)
        set(OPTIONS ${OPTIONS} /permissive-)
    endif()
    set(OPTIONS ${OPTIONS} /wd4702) # Disable warning C4702: unreachable code

    check_cxx_compiler_flag(/std:c++14 HAS_CPP14_FLAG)
    check_cxx_compiler_flag(/std:c++17 HAS_CPP17_FLAG)
    check_cxx_compiler_flag(/std:c++20 HAS_CPP20_FLAG)
    check_cxx_compiler_flag(/std:c++latest HAS_CPPLATEST_FLAG)
    set(CXX11_STANDARD "")
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    set(CMAKE_VERBOSE_MAKEFILE ON)
    set(OPTIONS -Wall -Wextra -pedantic-errors -Werror)

    check_cxx_compiler_flag(-std=c++14 HAS_CPP14_FLAG)
    check_cxx_compiler_flag(-std=c++17 HAS_CPP17_FLAG)
    check_cxx_compiler_flag(-std=c++20 HAS_CPP20_FLAG)
endif()

function(configure_target target std)
    target_compile_options(${target} PRIVATE ${OPTIONS})
    target_link_libraries(${target} PRIVATE scope_guard::scope_guard)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(std)
        if(SCOPE_GUARD_MSVC_FRONTEND)
            target_compile_options(${target} PRIVATE /std:${std})
        else()
            target_compile_options(${target} PRIVATE -std=${std})
        endif()
    endif()
endfunction()

function(add_doctest target source std)
    add_executable(${target} ${source})
    configure_target(${target} "${std}")
    target_compile_definitions(${target} PRIVATE DOCTEST_CONFIG_USE_STD_HEADERS)
    target_include_directories(${target} PRIVATE 3rdparty/doctest)
    add_test(NAME ${target} COMMAND ${target})
endfunction()

function(add_config_test target scenario std)
    add_doctest(${target} config_test.cpp "${std}")
    target_compile_definitions(${target} PRIVATE ${scenario})
endfunction()

function(add_program_test target source std)
    add_executable(${target} ${source})
    configure_target(${target} "${std}")
    add_test(NAME ${target} COMMAND ${target})
endfunction()

function(add_runtime_failure_test target scenario std expected_diagnostic)
    add_executable(${target} failure_test.cpp)
    configure_target(${target} "${std}")
    target_compile_definitions(${target} PRIVATE ${scenario})
    add_test(NAME ${target} COMMAND ${CMAKE_COMMAND} "-DTEST_PROGRAM=$<TARGET_FILE:${target}>" "-DEXPECTED_DIAGNOSTIC=${expected_diagnostic}" -P "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.cmake")
endfunction()

function(add_compile_failure_test target scenario std expected_diagnostic)
    add_executable(${target} EXCLUDE_FROM_ALL failure_test.cpp)
    configure_target(${target} "${std}")
    target_compile_definitions(${target} PRIVATE ${scenario})
    add_test(NAME ${target} COMMAND ${CMAKE_COMMAND} "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DBUILD_TARGET=${target}" "-DBUILD_CONFIG=$<CONFIG>" "-DEXPECTED_DIAGNOSTIC=${expected_diagnostic}" -P "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.cmake")
    set_tests_properties(${target} PROPERTIES RESOURCE_LOCK scope_guard_build_tree)
endfunction()
