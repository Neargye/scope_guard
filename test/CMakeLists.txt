# Compile features also cover pre-standard spellings such as GCC 9 -std=c++2a.
set(SUPPORTED_STANDARDS)
foreach(std IN LISTS STANDARDS)
    if(std STREQUAL "latest")
        check_cxx_compiler_flag(/std:c++latest HAS_CPPLATEST_FLAG)
        if(HAS_CPPLATEST_FLAG)
            list(APPEND SUPPORTED_STANDARDS ${std})
        endif()
    elseif(cxx_std_${std} IN_LIST CMAKE_CXX_COMPILE_FEATURES)
        list(APPEND SUPPORTED_STANDARDS ${std})
    endif()
endforeach()
if(NOT SUPPORTED_STANDARDS)
    message(STATUS "scope_guard tests skipped: no supported C++ standard for ${CMAKE_CXX_COMPILER_ID}")
    return()
endif()
list(GET SUPPORTED_STANDARDS 0 DEFAULT_STD)

# scope_guard_test(<name> <source> [DOCTEST] [STD <std>] [DEFINES ...] [OPTIONS ...]
#                  [COMPILE_ERROR <regex> | RUNTIME_ERROR <regex>])
function(scope_guard_test name source)
    cmake_parse_arguments(PARSE_ARGV 2 arg "DOCTEST" "STD;COMPILE_ERROR;RUNTIME_ERROR" "DEFINES;OPTIONS")
    set(target ${PROJECT_NAME}-${name}.t)
    if(NOT arg_STD)
        set(arg_STD ${DEFAULT_STD})
    endif()

    if(arg_COMPILE_ERROR)
        add_executable(${target} EXCLUDE_FROM_ALL ${source})
    else()
        add_executable(${target} ${source})
    endif()
    target_link_libraries(${target} PRIVATE scope_guard::scope_guard)
    if(arg_STD STREQUAL "latest")
        set_property(TARGET ${target} PROPERTY CXX_STANDARD)
        target_compile_options(${target} PRIVATE /std:c++latest)
    else()
        set_target_properties(${target} PROPERTIES CXX_STANDARD ${arg_STD} CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    endif()
    target_compile_options(${target} PRIVATE ${OPTIONS} ${arg_OPTIONS})
    target_compile_definitions(${target} PRIVATE ${arg_DEFINES})
    if(arg_DOCTEST)
        target_include_directories(${target} PRIVATE 3rdparty/doctest)
        target_compile_definitions(${target} PRIVATE DOCTEST_CONFIG_USE_STD_HEADERS)
    endif()

    if(arg_COMPILE_ERROR)
        add_test(NAME ${target} COMMAND ${CMAKE_COMMAND} "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DBUILD_TARGET=${target}" "-DBUILD_CONFIG=$<CONFIG>"
                 "-DEXPECTED_DIAGNOSTIC=${arg_COMPILE_ERROR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.cmake")
        set_tests_properties(${target} PROPERTIES RESOURCE_LOCK scope_guard_build_tree)
    elseif(arg_RUNTIME_ERROR)
        add_test(NAME ${target} COMMAND ${CMAKE_COMMAND} "-DTEST_PROGRAM=$<TARGET_FILE:${target}>"
                 "-DEXPECTED_DIAGNOSTIC=${arg_RUNTIME_ERROR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.cmake")
    else()
        add_test(NAME ${target} COMMAND ${target})
    endif()
endfunction()
